pub fn impls_trait_static<T, I>() -> bool
where
    T: 'static,
    I: 'static + ?Sized,
Expand description

Check if type T was registered to be casted to trait I using type_to_trait.

Example:

trait Trait {}
struct S;
impl Trait for S {}
type_to_trait!(S, Trait);
assert!(impls_trait_static::<S, dyn Trait>());
struct S2;
assert!(!impls_trait_static::<S2, dyn Trait>());
